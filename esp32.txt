Part 1: Wi-Fi Scanning
Step 5: Open WiFi Scan Example
Go to
File
    Examples
        WiFi
            WiFiScan

or paste this code:
#include "WiFi.h"

void setup() {
  Serial.begin(115200);

  WiFi.mode(WIFI_STA);
  WiFi.disconnect();

  delay(1000);

  Serial.println("Scanning...");
}

void loop() {

  int n = WiFi.scanNetworks();

  if (n == 0) {
    Serial.println("No networks found");
  }
  else {

    Serial.printf("%d networks found\n", n);

    for (int i = 0; i < n; i++) {

      Serial.print(i + 1);
      Serial.print(": ");

      Serial.print(WiFi.SSID(i));

      Serial.print(" RSSI:");

      Serial.print(WiFi.RSSI(i));

      Serial.print(" dBm Channel:");

      Serial.println(WiFi.channel(i));
    }
  }

  Serial.println();

  delay(5000);
}

Part 2: Bluetooth Scanning (BLE)
The ESP32 can scan for nearby Bluetooth Low Energy (BLE) devices that are advertising.
Step 8: Open BLE Scan Example
Go to
File
    Examples
        ESP32 BLE Arduino
            BLE Scan

If you don't see this example, ensure you have the latest ESP32 board package installed.
Use this example code:
#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEScan.h>

BLEScan* pBLEScan;

void setup() {
  Serial.begin(115200);

  BLEDevice::init("");

  pBLEScan = BLEDevice::getScan();

  pBLEScan->setActiveScan(true);
}

void loop() {

  BLEScanResults* results = pBLEScan->start(5, false);

  Serial.print("Devices Found: ");

  Serial.println(results->getCount());

  pBLEScan->clearResults();

  delay(3000);
}

Try:
●	Hold the BOOT button on the ESP32 while clicking Upload.
●	Release BOOT when "Connecting..." appears in the Arduino IDE.
If no COM port appears:
●	Install the correct USB-to-serial driver for your board (commonly CP210x or CH340, depending on the USB chip).
●	Try a different USB cable; some cables provide power only and do not carry data.

●	Verify the baud rate matches the sketch (e.g., 115200).

If you want to scan for nearby Bluetooth Low Energy (BLE) devices and print their advertised names, you can use the following Arduino sketch on an ESP32. It will list the names of nearby BLE devices that are advertising a name. Note that many BLE devices do not advertise a name, so some entries may show "(No Name)".
#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEScan.h>
#include <BLEAdvertisedDevice.h>

BLEScan* pBLEScan;

class MyAdvertisedDeviceCallbacks : public BLEAdvertisedDeviceCallbacks {
  void onResult(BLEAdvertisedDevice advertisedDevice) {

    Serial.println("----------------------");

    if (advertisedDevice.haveName()) {
      Serial.print("Device Name: ");
      Serial.println(advertisedDevice.getName().c_str());
    } else {
      Serial.println("Device Name: (No Name)");
    }

    Serial.print("MAC Address: ");
    Serial.println(advertisedDevice.getAddress().toString().c_str());

    Serial.print("RSSI: ");
    Serial.print(advertisedDevice.getRSSI());
    Serial.println(" dBm");
  }
};

void setup() {
  Serial.begin(115200);

  Serial.println("Starting BLE Scan...");

  BLEDevice::init("");

  pBLEScan = BLEDevice::getScan();
  pBLEScan->setAdvertisedDeviceCallbacks(new MyAdvertisedDeviceCallbacks());

  pBLEScan->setActiveScan(true);
  pBLEScan->setInterval(100);
  pBLEScan->setWindow(99);
}

void loop() {
  Serial.println("\nScanning for 5 seconds...");

  pBLEScan->start(5, false);
  pBLEScan->clearResults();

  delay(2000);
}

Important Notes
●	This code scans for Bluetooth Low Energy (BLE) devices only.
●	It can only display the advertised name if the device includes one in its BLE advertisements.
●	Many phones, laptops, headphones, and watches may not advertise a name continuously, or they may randomize their Bluetooth address for privacy.
If instead you mean Classic Bluetooth device discovery (such as finding nearby phones, laptops, or speakers that are discoverable via Bluetooth Classic), let me know. The code is different and support depends on the specific ESP32 model.

BLUETOOTH PROGRAM


#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEScan.h>
#include <BLEAdvertisedDevice.h>

int scanTime = 5; // In seconds
BLEScan* pBLEScan;

class MyAdvertisedDeviceCallbacks: public BLEAdvertisedDeviceCallbacks {
    void onResult(BLEAdvertisedDevice advertisedDevice) {
      // 1. Get the device name
      String displayName  = advertisedDevice.getName();

     // 2. If the name is blank, assign a placeholder
      if (displayName .length() == 0) {
        displayName = "[Unknown Device]";
      }
           
      // 3. Get the MAC Address
      String macAddress = advertisedDevice.getAddress().toString().c_str();

      // 4. Print everything if RSSI is available
      if (advertisedDevice.haveRSSI()) {
        Serial.print("Name: ");
        Serial.print(displayName);
        Serial.print(" | MAC: ");
        Serial.print(macAddress);
        Serial.print(" | RSSI: ");
        Serial.print(advertisedDevice.getRSSI());
        Serial.println(" dBm");
      }
    }
};

void setup() {
  Serial.begin(115200);
  Serial.println("Scanning...");

  BLEDevice::init("");
  pBLEScan = BLEDevice::getScan(); //create new scan
  pBLEScan->setAdvertisedDeviceCallbacks(new MyAdvertisedDeviceCallbacks());
  pBLEScan->setActiveScan(true); //active scan uses more power but gets more results
  pBLEScan->setInterval(100);
  pBLEScan->setWindow(99);  // must be less or equal to setInterval
}

void loop() {
  // Put the scan to work for the duration set in scanTime
  BLEScanResults* foundDevices = pBLEScan->start(scanTime, false);
  Serial.println("Scan done!");
  pBLEScan->clearResults();   // delete results fromBLEScan buffer to release memory
  delay(2000);
}
